{{/* Overrides PaperMod's twitter_cards partial. The theme default falls back to `summary` (a small, text-only card) whenever a post has no cover image — which was every post here. Social cards are generated per-slug by scripts/ogimage.py, so we always have a large image to point at. Resolution order: post cover.image -> /images/og/.png -> default.png */}} {{- $ogImage := "" }} {{- /* True when the site-wide fallback card was chosen: its alt text has to describe that card, which does not show this page's title. */}} {{- $isDefaultCard := false }} {{- with .Params.cover.image }} {{- if (ne $.Params.cover.relative true) }} {{- $ogImage = . | absURL }} {{- else }} {{- $ogImage = (path.Join $.RelPermalink .) | absURL }} {{- end }} {{- end }} {{- if not $ogImage }} {{- with .File }} {{- $candidate := printf "static/images/og/%s.png" .BaseFileName }} {{- if os.FileExists $candidate }} {{- $ogImage = (printf "/images/og/%s.png" .BaseFileName) | absURL }} {{- end }} {{- end }} {{- end }} {{- if not $ogImage }} {{- if os.FileExists "static/images/og/default.png" }} {{- $ogImage = "/images/og/default.png" | absURL }} {{- $isDefaultCard = true }} {{- end }} {{- end }} {{- if $ogImage }} {{- /* A per-slug card (scripts/ogimage.py) shows the page's title, so the title is its alt text. The fallback card shows the site headline instead; giving it the page's title described About, the section lists and every tag page as a card that says something else. On the homepage .Title is the site title, which used to render "Zhanyl Abdybaeva — Zhanyl Abdybaeva". */}} {{- $alt := printf "%s — Zhanyl Abdybaeva" ((or .Title site.Title) | plainify) }} {{- if $isDefaultCard }} {{- $alt = site.Params.defaultCardAlt | default site.Params.homeTitle | default site.Title | plainify }} {{- else if .IsHome }} {{- $alt = site.Params.homeTitle | default site.Title | plainify }} {{- end }} {{- else }} {{- end }} {{- $twTitle := or .Title site.Title site.Params.title }} {{- if .IsHome }}{{ $twTitle = site.Params.homeTitle | default $twTitle }}{{ end }} {{- with $twTitle | plainify }} {{- end }} {{- with or .Description .Summary site.Params.description | plainify | htmlUnescape }} {{- end }} {{- with site.Params.social }} {{- if reflect.IsMap . }} {{- with .twitter }} {{- $handle := . }} {{- if not (strings.HasPrefix . "@") }}{{ $handle = printf "@%v" . }}{{ end }} {{- end }} {{- end }} {{- end }}
